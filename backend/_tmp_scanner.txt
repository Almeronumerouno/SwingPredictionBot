55|def _count_gates_passed(gates: dict | None) -> int:
56|    """Count how many of the 4 gates (below, density, min_heavy, above_ma) are True."""
57|    if not gates:
58|        return 0
59|    return sum(1 for k in ("below", "density", "min_heavy", "above_ma") if gates.get(k))

62|def _fetch_and_check_one(
63|    code: str,
64|    name: str,
65|    daily_data: dict,
66|    target_date: Optional[str] = None,
67|    bars=None,
68|) -> Optional[ReadyToFlyEntry]:
69|    """Fetch history for one stock and run accumulation detection.
70|    bars bolehdiberikan dari luar (dipakai scan_all agar 1 fetch dipakai 3 analisis)."""
71|    try:
72|        if bars is None:
73|            bars = fetch_trading_info(
74|                code,
75|                length=config.RECOVERY_HISTORY_LOOKBACK_DAYS,
76|                target_date=target_date,
77|            )
78|        if len(bars) < config.RECOVERY_MIN_BARS:
79|            return None
80|
81|        accum = detect_accumulation(bars)
82|
83|        # Determine status
84|        is_ready = accum.get("ready_to_fly", False)
85|        gates = accum.get("gates")
86|        gates_passed = _count_gates_passed(gates)
87|
88|        # Keep if ready OR almost ready (>= 3 of 4 gates passed)
89|        if not is_ready and gates_passed < 3:
90|            return None
91|
92|        status = "ready" if is_ready else "almost"
93|
94|        close_arr = [b.close for b in bars]
95|        close_now = float(close_arr[-1]) if close_arr else 0.0
96|
97|        prev = float(daily_data.get("Previous", 0) or 0)
98|        c = float(daily_data.get("Close", close_now) or close_now)
99|
100|        # Mode yahoo (scan_all dengan force_source="yahoo") tidak punya snapshot
101|        # IDX: daily_data cuma {StockCode, StockName}, jadi Previous/Close = 0 dan
102|        # pct_change bakal 0.00% semua. Fallback: hitung dari dua bar terakhir
103|        # Yahoo (harga mentah = % change riil pasar, konsisten dgn snapshot IDX).
104|        if not prev and len(bars) >= 2:
105|            raw = [(getattr(b, "raw_close", None) or b.close) for b in bars]
106|            prev, c = float(raw[-2]), float(raw[-1])
107|
108|        pct = ((c - prev) / prev * 100.0) if prev else 0.0
109|
110|        return ReadyToFlyEntry(
111|            code=code,
112|            name=name,
113|            close=c,
114|            pct_change=pct,
115|            status=status,
116|            density_pct=accum.get("density_pct"),
117|            k_heavy=accum.get("k_heavy", 0),

122|            sma20=accum.get("sma20"),

125|            gates=gates,
126|            note=accum.get("note"),
